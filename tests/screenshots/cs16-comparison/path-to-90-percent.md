  let whiteoutDuration = 0;

    const brightness = await this.page.evaluate(() => {
      const canvas = document.querySelector('canvas');
      // 分析canvas平均亮度
      return calculateAverageBrightness(canvas);
    });
    if (brightness < 200) {  // 白屏阈值
      whiteoutDuration = (Date.now() - startTime) / 1000;
      break;
    }
    await this.page.waitForTimeout(50);
  }

}
