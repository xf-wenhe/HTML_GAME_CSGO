```javascript
// 修改测试脚本，增加调试和重试机制
async waitForState(predicate, timeout = 30000) {  // 增加超时时间
  const start = Date.now();
  let lastError = null;

  while (Date.now() - start < timeout) {
    try {
      const state = await this.getState();
      if (predicate(state)) {
        console.log('✓ 状态条件满足');
        return state;
      }
    } catch (e) {
      lastError = e;
      console.log('状态获取失败，重试中...', e.message);
    }
    await this.page.waitForTimeout(500);  // 增加重试间隔
  }

  throw new Error(`Timeout: ${lastError?.message || 'State not reached'}`);
}
```
